import re
src=open("ledger/unasked-ledger.md").read()
blocks=re.findall(r"^ID: (UQ-\S+)\n(.*?)(?=\n---\n|\Z)", src, re.S|re.M)
rows=[]
for eid,body in blocks:
    def f(name):
        m=re.search(rf"^{re.escape(name)}:\s*(.*?)(?=\n[A-Z][^\n:]{{2,60}}:|\Z)", body, re.S|re.M)
        return " ".join(m.group(1).split()) if m else ""
    status=f("Status").split("—")[0].split("(")[0].strip() or "?"
    dom=f("Domain(s)").split(";")[0].strip()[:60]
    q=re.sub(r"^\[[^\]]*\]\s*","",f("Question (canonical)"))[:200]
    rows.append(f"| {eid} | {status} | {dom} | {q} |")
out=["# Ledger index","","One line per entry, newest last. Generated from `unasked-ledger.md`, which stays the append-only record.",
 "Read this file for the Gate 2 duplicate check, then open the full entry in `unasked-ledger.md` only for candidates that look close.","",
 "| ID | Status | Domain | Question (truncated) |","|---|---|---|---|"]+rows+[""]
open("ledger/INDEX.md","w").write("\n".join(out))
